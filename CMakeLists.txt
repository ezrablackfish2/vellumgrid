cmake_minimum_required(VERSION 3.16)
project(vellumgrid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB VELLUMGRID_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(vellumgrid STATIC ${VELLUMGRID_SOURCES})
target_include_directories(vellumgrid PUBLIC include)
target_compile_options(vellumgrid PRIVATE -Wall -Wextra -Wpedantic)
add_executable(vellumgrid_smoke tests/smoke_tests.cpp)
target_link_libraries(vellumgrid_smoke PRIVATE vellumgrid)
enable_testing()
add_test(NAME smoke COMMAND vellumgrid_smoke)
